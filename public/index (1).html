const logEl = document.getElementById('log');
const statusEl = document.getElementById('status');
const form = document.getElementById('cmd-form');
const input = document.getElementById('cmd-input');
async function refreshStatus() {
try {
const res = await fetch('/health');
const data = await res.json();
statusEl.innerHTML =
'<span class="dot ' + (data.anthropicConfigured ? 'on' : 'off') + '"></span>Claude: ' +
(data.anthropicConfigured ? 'live' : 'dry-run') +
' &nbsp; <span class="dot ' + (data.ghlConfigured ? 'on' : 'off') + '"></span>GoHighLevel: ' +
(data.ghlConfigured ? (data.ghlSendMode === 'live' ? 'live (sending)' : 'connected (review mode - nothing sends)') : 'dry-run') +
' &nbsp; brand: ' + data.defaultBrain;
} catch {
statusEl.textContent = 'Cannot reach server.';
}
}
function redirectIfSignedOut(res) {
if (res.status === 401) {
window.location.href = '/login.html';
return true;
}
return false;
}
async function refreshLeads() {
try {
const res = await fetch('/api/leads');
if (redirectIfSignedOut(res)) return;
const data = await res.json();
const tbody = document.querySelector('#leads-table tbody');
tbody.innerHTML = '';
// Older copies of a re-sent contact are hidden; names are escaped
// because they arrive from outside (web forms, GoHighLevel).
(data.leads || []).filter((l) => l.status !== 'superseded').forEach((l) => {
const tr = document.createElement('tr');
tr.innerHTML = '<td>' + escapeHtml([l.first_name, l.last_name].filter(Boolean).join(' ')) + '</td>' +
'<td>' + escapeHtml(STATUS_LABELS[l.status] || l.status || '') + '</td><td>' + escapeHtml(l.source || '') + '</td>';
tbody.appendChild(tr);
});
} catch {}
}
function addLogEntry(instruction, result) {
const div = document.createElement('div');
div.className = 'entry';
const badge = result.dryRun ? '<span class="badge dry">DRY RUN</span>' : '<span class="badge">' + result.intent + '</span>';
const r = result.response || {};
let body;
if (Array.isArray(r.drafts)) {
body = '';
if (r.figuresToVerify && r.figuresToVerify.length) {
body += '<div class="verify"><strong>Check before posting:</strong> these figures are not on your approved-facts list, so they may be made up: ' +
r.figuresToVerify.map(escapeHtml).join(', ') + '. Replace them with a sourced number or remove them.</div>';
}
r.drafts.forEach((d, i) => {
body += '<div class="draft-head">Draft ' + (i + 1) + '</div>' +
'<div class="draft"><button type="button" class="copy">Copy</button>' + escapeHtml(d) + '</div>';
});
if (r.notes) body += '<div class="draft-notes"><strong>Notes:</strong> ' + escapeHtml(r.notes) + '</div>';
} else if (Array.isArray(r.events)) {
body = r.events.length ? r.events.map(activityCard).join('') : '<div class="card">No activity yet.</div>';
body += rawBlock(r);
} else if (Array.isArray(r.leads)) {
body = '<table class="mini"><thead><tr><th>Name</th><th>Status</th><th>Source</th><th>Added</th></tr></thead><tbody>' +
r.leads.map((l) => '<tr><td>' + escapeHtml(((l.first_name || '') + ' ' + (l.last_name || '')).trim() || '(no name)') +
'</td><td>' + escapeHtml(STATUS_LABELS[l.status] || l.status || '') + '</td><td>' + escapeHtml(l.source || '') +
'</td><td>' + escapeHtml(when(l.created_at)) + '</td></tr>').join('') + '</tbody></table>';
} else {
body = '<pre>' + escapeHtml(JSON.stringify(r, null, 2)) + '</pre>';
}
div.innerHTML = '<div class="instr">' + escapeHtml(instruction) + badge + '</div>' + body;
div.querySelectorAll('.copy').forEach((btn) => {
btn.addEventListener('click', () => {
const text = btn.parentElement.textContent.replace(/^Copy/, '');
navigator.clipboard.writeText(text).then(() => { btn.textContent = 'Copied'; setTimeout(() => (btn.textContent = 'Copy'), 1500); });
});
});
logEl.prepend(div);
}
const AGENT_LABELS = { leadQualifier: 'New lead', followUp: 'Follow-up', appointmentSetter: 'Appointment', contentDrafter: 'Content drafts', ghlWebhook: 'GoHighLevel event' };
const ACTION_LABELS = {
qualified: ['Good fit - reply drafted', 'good'],
needs_human: ['Needs you', 'you'],
disqualified: ['Not a fit', 'no'],
blocked_compliance: ['Held for you - compliance', 'you'],
blocked_sms_compliance: ['Held for you - compliance', 'you'],
draft: ['Drafted', 'info'],
};
const STATUS_LABELS = { new: 'New', active: 'Active', needs_human: 'Needs you', disqualified: 'Not a fit', superseded: 'Replaced by newer copy', opted_out: 'Opted out', booked: 'Booked', won: 'Client' };
// Database times are UTC ("2026-10-02 13:47:39"); show them in local time.
function when(ts) {
if (!ts) return '';
const d = new Date(String(ts).replace(' ', 'T') + 'Z');
return isNaN(d) ? String(ts) : d.toLocaleString([], { month: 'short', day: 'numeric', hour: 'numeric', minute: '2-digit' });
}
function activityCard(e) {
const d = (e.detail && typeof e.detail === 'object') ? e.detail : {};
const p = d.parsed || d;
let [label, kind] = ACTION_LABELS[e.action] || [String(e.action || '').replace(/_/g, ' '), 'info'];
if (/^sent_attempt_(\d+)/.test(e.action || '')) {
const n = e.action.match(/\d+/)[0];
const held = d.sendResult && d.sendResult.held;
label = 'Follow-up #' + n + (held ? ' - saved as draft' : ' - sent');
kind = held ? 'info' : 'good';
}
const who = e.leadName || (e.agent === 'contentDrafter' ? 'Content' : 'General');
let html = '<div class="card"><div class="top"><span class="who">' + escapeHtml(who) + ' &middot; ' +
escapeHtml(AGENT_LABELS[e.agent] || e.agent) + '</span><span class="when">' + escapeHtml(when(e.created_at)) + '</span></div>' +
'<div><span class="pill ' + kind + '">' + escapeHtml(label) + '</span>' + (e.dry_run ? ' <span class="pill no">dry run</span>' : '') + '</div>';
const why = p.reason || d.reason;
if (why) html += '<div class="why"><strong>Why:</strong> ' + escapeHtml(why) + '</div>';
if (d.instruction) html += '<div class="why"><strong>Asked for:</strong> ' + escapeHtml(d.instruction) + '</div>';
const msg = p.reply || p.message || p.draftForDrGrove;
if (msg) html += '<div class="reply">' + escapeHtml(msg) + '</div>';
if (p.suggestedTag) html += '<div class="tag">Suggested tag: ' + escapeHtml(p.suggestedTag) + '</div>';
return html + '</div>';
}
function rawBlock(r) {
return '<details class="raw"><summary>Show raw data</summary><pre>' + escapeHtml(JSON.stringify(r, null, 2)) + '</pre></details>';
}
function escapeHtml(s) {
return String(s).replace(/[&<>"']/g, (c) => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}
form.addEventListener('submit', async (e) => {
e.preventDefault();
const instruction = input.value.trim();
if (!instruction) return;
input.disabled = true;
try {
const res = await fetch('/command', {
method: 'POST',
headers: { 'content-type': 'application/json' },
body: JSON.stringify({ instruction }),
});
if (redirectIfSignedOut(res)) return;
const result = await res.json();
addLogEntry(instruction, result);
input.value = '';
refreshLeads();
} catch (err) {
addLogEntry(instruction, { response: { error: String(err) } });
} finally {
input.disabled = false;
input.focus();
}
});
refreshStatus();
refreshLeads();
</script>
